import pandas as pd, numpy as np, importlib.util
s=importlib.util.spec_from_file_location('e','code/evaluation/evaluate_multi_model_ensemble.py');e=importlib.util.module_from_spec(s);s.loader.exec_module(e)
d=pd.read_csv('docs/manifests/ensemble_phase3_audit_summary.csv');ms=['MedNeXt_Phase3','Swin-UNETR_Phase3'];orgs=['stomach','duodenum','small_bowel','colon'];out=[]
for _,r in d.iterrows():
 od=[];inter=[]
 for o in orgs:
  vals=[r[f'{o}_dice_{m}'] for m in ms if not (r[f'{o}_betti_consensus']==0 and r[f'{o}_betti_diff_{m}']==0)]
  if vals:od.append(np.mean(vals))
  if not (r[f'{o}_betti_consensus']==0 and all(r[f'{o}_betti_diff_{m}']==0 for m in ms)):inter.append(r[f'{o}_inter_model_dice'])
 md=float(np.mean(od)) if od else 0;mi=float(np.mean(inter)) if inter else 0
 cat,_=e.triage_case(md,mi,r.max_betti_diff,r.mean_uncertainty,total_foreground_components=sum(r[f'{o}_betti_consensus'] for o in orgs))
 out.append(dict(subject_id=r.subject_id,old=r.triage_category,reaggregated=cat,old_dice=r.mean_consensus_dice,new_dice=md,old_inter=r.mean_inter_model_dice,new_inter=mi))
o=pd.DataFrame(out);o.to_csv('/tmp/mp_archive_reaggregate.csv',index=False)
print(o.reaggregated.value_counts());print(pd.crosstab(o.old,o.reaggregated));print(o[o.old!=o.reaggregated].head(10).to_string(index=False))
